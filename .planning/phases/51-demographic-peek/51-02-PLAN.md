    Add the following to the `AdLibraryContent` component in page.tsx:

    **1. Imports:**
    ```typescript
    import { DemographicPeek } from './components/demographic-peek';
    import { normalizeDemographicsJson, NormalizedDemographics } from '@/lib/demographics-normalizer';
    ```

    **2. State (add after existing state declarations, around line 85):**
    ```typescript
    const [brandDemographics, setBrandDemographics] = useState<NormalizedDemographics | null>(null);
    const [demoCollapsed, setDemoCollapsed] = useState<boolean>(() => {
      if (typeof window !== 'undefined') {
        return localStorage.getItem('demographicPeekCollapsed') === 'true';
      }
      return false;
    });
    ```

    **3. Collapse toggle handler (persists to localStorage):**
    ```typescript
    const toggleDemoCollapsed = useCallback(() => {
      setDemoCollapsed(prev => {
        const next = !prev;
        localStorage.setItem('demographicPeekCollapsed', String(next));
        return next;
      });
    }, []);
    ```

    **4. useEffect to fetch demographics when brandFilter changes:**
    ```typescript
    useEffect(() => {
      if (!brandFilter) {
        setBrandDemographics(null);
        return;
      }
      let cancelled = false;
      fetch(`/api/ad-library/brands/${brandFilter}`)
        .then(res => res.ok ? res.json() : null)
        .then(data => {
          if (cancelled) return;
          if (data?.brand?.demographicsJson) {
            const normalized = normalizeDemographicsJson(data.brand.demographicsJson);
            setBrandDemographics(normalized);
          } else {
            setBrandDemographics(null);
          }
        })
        .catch(() => {
          if (!cancelled) setBrandDemographics(null);
        });
      return () => { cancelled = true; };
    }, [brandFilter]);
    ```

    This effect is INDEPENDENT from the ads fetch. It uses the brand detail endpoint which already returns full brand data including the newly-added demographicsJson field. The `cancelled` flag prevents stale updates if brandFilter changes rapidly.

    **5. Render DemographicPeek between FilterBar and the Ads Grid section (after the FilterBar closing tag, before the `<section>` with "Browse Ads"):**
    ```tsx
    {brandDemographics && (
      <DemographicPeek
        demographics={brandDemographics}
        darkMode={darkMode}
        collapsed={demoCollapsed}
        onToggleCollapse={toggleDemoCollapsed}
      />
    )}
    ```

    **6. Reset demographics when filters are cleared:** In the `clearAllFilters` function, add `setBrandDemographics(null);` (the useEffect will handle it when brandFilter becomes '', but explicitly clearing prevents a flash of stale data).

    **Important:** Do NOT make the demographics fetch block the ads grid. They are completely independent fetches.
